    96→    96→    firstName: '',

   108→   108→  const [photoPreview, setPhotoPreview] = useState('');
   109→   109→  const [photoUploading, setPhotoUploading] = useState(false);
   110→   110→  const [photoError, setPhotoError] = useState('');
   111→   111→
   112→   112→  // REWARD-FEATURE: récompense proposée en cas de perte
   113→   113→  const [reward, setReward] = useState('');
   114→   114→

   116→   116→  useEffect(() => {
   117→   117→    if ((countryCode && countryCode !== 'FR') || !phoneCountry) {
   118→   118→      setPhoneCountry(countryCode);
   119→   119→    }
   120→   120→  }, [countryCode]);

   122→   122→  // 🔒 Référence absente → activation impossible

   124→   124→

   126→   126→  const compressAndUpload = async (file: File) => {
   127→   127→    setPhotoError('');
   128→   128→    setPhotoUploading(true);
   129→   129→    try {
   130→   130→      const dataUrl = await new Promise<string>((resolve, reject) => {
   131→   131→        const reader = new FileReader();
   132→   132→        reader.onload = () => resolve(reader.result as string);
   133→   133→        reader.onerror = reject;
   134→   134→        reader.readAsDataURL(file);
   135→   135→      });
   136→   136→
   137→   137→      const img = new window.Image();
   138→   138→      await new Promise<void>((resolve, reject) => {
   139→   139→        img.onload = () => resolve();
   140→   140→        img.onerror = () => reject(new Error('image'));
   141→   141→        img.src = dataUrl;
   142→   142→      });
   143→   143→
   144→   144→      const maxDim = 1200;
   145→   145→      let { width, height } = img;
   146→   146→      if (width > maxDim || height > maxDim) {
   147→   147→        const ratio = Math.min(maxDim / width, maxDim / height);
   148→   148→        width = Math.round(width * ratio);
   149→   149→        height = Math.round(height * ratio);
   150→   150→      }
   151→   151→      const canvas = document.createElement('canvas');
   152→   152→      canvas.width = width;
   153→   153→      canvas.height = height;
   154→   154→      const ctx = canvas.getContext('2d');
   155→   155→      if (!ctx) throw new Error('canvas');
   156→   156→      ctx.drawImage(img, 0, 0, width, height);
   157→   157→
   158→   158→      const blob = await new Promise<Blob | null>((resolve) => canvas.toBlob(resolve, 'image/jpeg', 0.8));
   159→   159→      const compressed = blob ?? file;
   160→   160→
   161→   161→      const fd = new FormData();
   162→   162→      fd.append('file', compressed, 'photo-valise.jpg');
   163→   163→      const res = await fetch('/api/baggage-photo/upload', { method: 'POST', body: fd });
   164→   164→      if (!res.ok) throw new Error('upload');
   165→   165→      const data = await res.json();
   166→   166→
   167→   167→      setPhotoPath(data.photoPath || '');
   168→   168→      setPhotoPreview(canvas.toDataURL('image/jpeg', 0.6));
   169→   169→    } catch {
   170→   170→      setPhotoError(t('inscrire.photo_error'));
   171→   171→    } finally {
   172→   172→      setPhotoUploading(false);
   173→   173→    }
   174→   174→  };

   178→   178→    e.target.value = ''; // permet de re-sélectionner le même fichier

   180→   180→  };

   192→   192→          travelerFirstName: formData.firstName,
   193→   193→          travelerLastName: formData.lastName,
   194→   194→          whatsappOwner: formData.whatsapp,
   195→   195→          // Plus de sélection de transport côté UI — l'API applique son défaut ('flight',
   196→   196→          // cohérent avec les références voyageur VOL26-)
   197→   197→          transportMode: 'flight',
   198→   198→          destination: formData.destination,
   199→   199→          departureDate: formData.departureDate || undefined,
   200→   200→          departureTime: formData.departureTime || undefined,
   201→   201→          // PHOTO + REWARD FEATURE
   202→   202→          photoPath: photoPath || undefined,
   203→   203→          reward: reward.trim() || undefined,
   204→   204→        }),
   205→   205→      });
   206→   206→
   207→   207→      if (response.ok) {
   208→   208→        const data = await response.json();
   209→   209→        sessionStorage.setItem(
   210→   210→          'activationData',
   211→   211→          JSON.stringify({
   212→   212→            reference: formData.reference,

   214→   214→            lastName: formData.lastName,
   215→   215→            whatsapp: formData.whatsapp,

   230→   230→      }
